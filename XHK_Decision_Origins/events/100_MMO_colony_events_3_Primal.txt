
namespace = colony

#Gatekeeper - Colony Events
carrier_event = {
	id = colony_mod.161
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			AND = {
				has_modifier = strong_magnetic_field
				NOT = { has_carrier_flag = magnetic_mystery_solved }
			}
			has_modifier = asteroid_impacts
			has_modifier = irradiated_planet
			has_modifier = titanic_life
			has_modifier = unstable_tectonics
			has_modifier = asteroid_belt
			has_modifier = atmospheric_hallucinogen
		}
		owner = { is_wilderness_empire = no }
	}

	immediate = {
		if = {
			limit = {
				has_modifier = strong_magnetic_field
				has_modifier = "strong_magnetic_field"
				NOT = { has_modifier = "magnetic_research_boost" }
				NOT = { has_modifier = "magnetic_disruption" }
				has_ground_combat = no
				pop_amount > 0
			}
			random_list = {
				50 = { carrier_event = { id = colony_mod.16 days = 30 random = 90 } }
				50 = { carrier_event = { id = colony_mod.17 days = 30 random = 90 } }
			}
			break = yes
		}
		#if = {
		#	limit = {
		#		has_modifier = asteroid_impacts
		#		NOT = { has_carrier_flag = asteroid_defences_colony_flag }
		#		pop_amount > 100
		#		NOT = { has_modifier = "recent_impact" }
		#	}
		#	random_list = {
		#		50 = { carrier_event = { id = colony_mod.12 days = 30 random = 90 } }
		#		50 = { carrier_event = { id = colony_mod.14 days = 30 random = 90 } }
		#	}
		#	break = yes
		#}

		if = {
			limit = {
				has_modifier = irradiated_planet
				NOT = { has_carrier_flag = irradiated_colony_flag }
			}
			carrier_event = { id = colony_mod.1 days = 30 random = 90 }
			break = yes
		}

		if = {
			limit = {
				has_modifier = titanic_life
				NOT = { has_carrier_flag = titanic_life_colony_flag }
			}
			if = {
				limit = {
					owner = { 
						OR = {
							has_origin = origin_primal_calling
							has_country_flag = decision_primal_calling
						}
					}
				}
				carrier_event = { id = grand_archive.1010 days = 30 random = 90 }
				break = yes
			}
			else_if = {
				limit = {
					NOT = { owner = { has_ethic = ethic_gestalt_consciousness } }
				}
				carrier_event = { id = colony_mod.2 days = 30 random = 90 }
				break = yes
			}
		}

		if = {
			limit = {
				has_modifier = unstable_tectonics
				NOT = { has_modifier = recent_quake }
			}
			carrier_event = { id = colony_mod.3 days = 30 random = 90 }
			break = yes
		}

		if = {
			limit = {
				has_modifier = "asteroid_belt"
				NOR = {
					has_modifier = "recent_comet"
					owner = { has_ethic = ethic_gestalt_consciousness }
				}
			}
			carrier_event = { id = colony_mod.4 days = 30 random = 90 }
			break = yes
		}

		if = {
			limit = {
				has_modifier = atmospheric_hallucinogen
				NOR = {
					has_carrier_flag = atmospheric_hallucinogen_colony_flag
					owner = { has_ethic = ethic_gestalt_consciousness }
				}
			}
			carrier_event = { id = colony_mod.5 days = 30 random = 90 }
			break = yes
		}
	}
}
