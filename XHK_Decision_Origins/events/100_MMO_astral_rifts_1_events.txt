	option = {
		exclusive_trigger = { explorer = {
				OR = {
					has_country_flag = decision_evolutionary_predators
					has_origin = origin_evolutionary_predators
				}
		} }
		name = astral_rift.1320.b.evopred
		explorer = {
			give_evolutionary_predator_situation_progress_and_dna = { AMOUNT = medium TYPE = toxoid }
		}
	}

	after = {
		explorer = {
			astral_thread_large_reward = yes
		}
		if = {
			limit = { has_astral_rift_flag = sample_rig_available }
			explorer = {
				add_modifier = {
					modifier = rift_samples_biology_modifier
					years = 10
				}
			}
		}
		else = {
			explorer = {
				add_modifier = {
					modifier = rift_samples_biology_modifier
					years = 5
				}
			}
		}
		finish_astral_rift = yes
	}
}
