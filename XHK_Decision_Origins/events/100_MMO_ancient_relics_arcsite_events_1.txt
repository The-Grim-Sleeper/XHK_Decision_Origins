# Zroni Introduction
# via on_survey = {} | this = surveying vessel
ship_event = {
	id = ancrel.1
	title = "ancrel.1.name"
	desc = "ancrel.1.desc"
	picture = GFX_evt_overgrown_city
	location = from
	show_sound = event_wind_ruins
	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		host_has_dlc = "Ancient Relics Story Pack"
		owner = {
			is_ai = no
			has_precursor_intro = no
		}
		solar_system = { has_star_flag = precursor_zroni_1 }
		NOT = { has_global_flag = zroni_started }
		FROM = {
			has_anomaly = no
			is_colony = no
			NOT = { exists = archaeological_site }
			NOT = { has_carrier_flag = suppress_archaeological_sites }
			habitable_planet = yes
		}
	}
	weight_multiplier = {
		factor = 1
		modifier = {
			factor = @origin_shoulders_multiplier
			owner = {
				OR = {
					has_country_flag = decision_evolutionary_predators
					has_origin = origin_evolutionary_predators
				}
				has_country_flag = origin_shoulders_closure
			}
		}
	}

	immediate = {
		owner = { set_country_flag = zroni_intro }
		set_global_flag = zroni_started
		solar_system = { set_star_flag = first_zroni_site@root.owner }
		from = { prevent_anomaly = yes }
	}

	option = {
		name = CURIOUS
		hidden_effect = {
			owner = { country_event = { id = story.5 days = 30 } }
		}
		from = {
			create_archaeological_site = zroni_digsite_1
			root.owner = {
				set_country_flag = zroni_arc_chain_@prev
			}
			add_deposit = d_zro_deposit_2
		}
		owner = {
			add_monthly_resource_mult = {
				resource = engineering_research
				value = @tier1researchreward
				min = @tier1researchmin
				max = @tier1researchmax
			}
		}
	}
}

			}

	option = {
		name = "ancrel.2072.a"
	}

	option = {
		name = ancrel.2702.b.evopred
		exclusive_trigger = {
			owner = {
				OR = {
					has_country_flag = decision_evolutionary_predators
					has_origin = origin_evolutionary_predators
				}
			}
		}
		owner = {
			give_evolutionary_predator_situation_progress_and_dna = { AMOUNT = large TYPE = arthropoid }
		}
	}

	after = {
		owner = {
			IF = {
				limit = {
					NOT = { has_technology = tech_colonization_3 }
					can_research_technology = tech_colonization_3
				}
				give_technology = { tech = tech_colonization_3 }
			}
			ELSE = {
				add_monthly_resource_mult = {
					resource = society_research
					value = @tier4researchreward
					min = @tier4researchmin
					max = @tier4researchmax
				}
			}
			if = {
				limit = {
					NOR = {
						has_technology = tech_archaeo_strike_crafts
						has_tech_option = tech_archaeo_strike_crafts
					}
				}
				add_research_option = tech_archaeo_strike_crafts
				add_tech_progress = {
					tech = tech_archaeo_strike_crafts
					progress = 0.25
				}
			}
			give_specimen = {
				key = fossilized_megastomach
				targets = { root.fromfrom.planet }
			}
		}
	}
}

fleet_event = {
	id = ancrel.4005
	title = ancrel.4005.name
	desc = ancrel.4005.desc
	picture = GFX_evt_underground_city
	show_sound = event_whispering
	location = from

	archaeology = yes

	is_triggered_only = yes

	immediate = {
		owner = { set_country_flag = lithoid_crystal_plant } # unlocks building_crystal_plant_2
		from.planet = { remove_modifier = lithoids_arc_site_minerals }
		from = { set_site_progress_locked = yes }
	}

	after = {
		from = { set_site_progress_locked = no }
		owner = {
			add_resource = {
				rare_crystals = 500
			}
			large_artifact_reward = yes
			give_specimen = {
				key = khalanka_crystallized_corpses
				targets = { root.from.planet }
			}
		}
		if = {
			limit = {
				owner = {
					NOT = {
						has_technology = tech_rare_crystals
					}
				}
			}
			owner = {
				give_technology = { tech = tech_rare_crystals } # Gives unique building
			}
		}
	}

	option = {
		name = REMARKABLE
	}

	option = {
		exclusive_trigger = { owner = {
			OR = {
				has_country_flag = decision_evolutionary_predators
				has_origin = origin_evolutionary_predators
			}
		} }
		name = ancrel.4005.b.evopred
		owner = {
			give_evolutionary_predator_situation_progress_and_dna = { AMOUNT = medium TYPE = lithoid }
		}
